ChatGPT의 말:

지금 코드에서는 forEach를 사용해 모든 이미지를 한 번에 순회하면서 sliderImg.src를 계속 덮어쓰고 있기 때문에, 마지막 이미지로만 보이게 됩니다.
너가 원하는 기능은 다음과 같습니다:

  • 처음에는 첫 번째 이미지가 출력됨

  • 다음 버튼(nextBtn)을 누를 때마다 다음 이미지로 변경됨

  • 마지막 이미지 이후에는 다시 처음 이미지로 돌아가거나 멈추게 할 수 있음

아래는 이를 구현한 정상 작동 코드 예시입니다:

javascript
let sliderImgs = [
    'img/sliderImg1.png',
    'img/sliderImg2.jpeg',
    'img/sliderImg3.jpg',
    'img/sliderImg4.jpg'
];

const sliderImg = document.getElementById("sliderImg");
const prewBtn = document.getElementById("prewBtn");
const nextBtn = document.getElementById("nextBtn");

let currentIndex = 0;

// 최초 이미지 출력
sliderImg.src = sliderImgs[currentIndex];

// 다음 버튼 클릭 이벤트
nextBtn.addEventListener("click", function() {
    currentIndex++;
    if (currentIndex >= sliderImgs.length) {
        currentIndex = 0;  // 다시 처음으로 순환하려면 이 줄을 유지
    }
    sliderImg.src = sliderImgs[currentIndex];
});

// 이전 버튼 클릭 이벤트 (선택 사항)
prewBtn.addEventListener("click", function() {
    currentIndex--;
    if (currentIndex < 0) {
        currentIndex = sliderImgs.length - 1;  // 마지막 이미지로 돌아가기
    }
    sliderImg.src = sliderImgs[currentIndex];
});

✅ 설명 요약:

  1. currentIndex 변수로 현재 보여주는 이미지의 인덱스를 추적

  2. nextBtn 클릭 시 currentIndex를 증가시키고 배열 범위를 벗어나면 다시 0으로

  3. prewBtn 클릭 시 currentIndex를 감소시키고 0보다 작아지면 마지막 인덱스로

📌 추가 팁 (선택 기능):

  • 이미지가 자연스럽게 전환되도록 하려면 CSS에서 transition을 주거나 fade 효과를 자바스크립트로 넣을 수 있어요.

  • 자동 슬라이드 쇼로 확장하려면 setInterval을 추가하면 됩니다.

✅ 2. 스크립트 위치 문제

🔍 문제 확인:

🛠 해결 방법:

JS 코드를 HTML의 <body> 끝에 넣거나, DOMContentLoaded 이벤트로 감싸주세요.

✅ 권장 방식 (가장 안전):

✅ 구현 목표

  1. sliderImgs 배열의 현재 currentIndex에 맞춰

  2. <li> 태그 중 해당 위치에만 .active 클래스 부여

  3. 나머지는 .active 클래스 제거

✅ 최종 JavaScript 코드 예시 (페이저 업데이트 포함):

javascript
document.addEventListener("DOMContentLoaded", function () {
    let sliderImgs = [
        'img/sliderImg1.png',
        'img/sliderImg2.jpeg',
        'img/sliderImg3.jpg',
        'img/sliderImg4.jpg'
    ];

    const sliderImg = document.getElementById("sliderImg");
    const prewBtn = document.getElementById("prewBtn");
    const nextBtn = document.getElementById("nextBtn");
    const pagerItems = document.querySelectorAll("#sliderPager li");

    let currentIndex = 0;

    // 이미지와 페이저 모두 업데이트하는 함수
    function updateSlider() {
        sliderImg.src = sliderImgs[currentIndex];

        // 모든 li에서 active 제거
        pagerItems.forEach((li, index) => {
            if (index === currentIndex) {
                li.classList.add("active");
            } else {
                li.classList.remove("active");
            }
        });
    }

    // 최초 출력
    updateSlider();

    // 다음 버튼
    nextBtn.addEventListener("click", function () {
        currentIndex = (currentIndex + 1) % sliderImgs.length;
        updateSlider();
    });

    // 이전 버튼
    prewBtn.addEventListener("click", function () {
        currentIndex = (currentIndex - 1 + sliderImgs.length) % sliderImgs.length;
        updateSlider();
    });

    // 페이저 버튼 클릭 시 해당 인덱스로 이동
    pagerItems.forEach((li, index) => {
        li.addEventListener("click", function () {
            currentIndex = index;
            updateSlider();
        });
    });
});

💡 추가 설명

기능설명
updateSlider()이미지와 .active 클래스 둘 다 업데이트
.forEach로 li 순회currentIndex에 맞는 li만 .active로 유지
(currentIndex + 1) % length슬라이드가 끝나면 처음으로 순환
pagerItems.forEach(...).addEventListener()페이지 버튼을 눌러도 이동 가능하게 함

🧪 테스트 시 유의사항

  • <li class="active">가 슬라이드 전환 시 자동 이동하는지 확인

  • <li> 안에 있는 <button> 클릭 시도 동작하는지 확인

  • CSS에 .active 스타일 정의가 되어 있어야 눈에 보이는 변화가 있습니다:

css
#sliderPager li.active button {
    background-color: black;
    color: white;
}

✅ 개선된 JS 코드: 스와이프 + 기존 기능 유지

javascript
document.addEventListener("DOMContentLoaded", function () {
    const sliderImgs = [
        'img/sliderImg1.png',
        'img/sliderImg2.jpeg',
        'img/sliderImg3.jpg',
        'img/sliderImg4.jpg'
    ];

    const sliderImg = document.getElementById("sliderImg");
    const prewBtn = document.getElementById("prewBtn");
    const nextBtn = document.getElementById("nextBtn");
    const pagerItems = document.querySelectorAll("#sliderPager li");

    let currentIndex = 0;
    let autoSlide = setInterval(goNextSlide, 3000);

    // 슬라이드 업데이트
    function updateSlider() {
        sliderImg.src = sliderImgs[currentIndex];
        pagerItems.forEach((li, i) => {
            li.classList.toggle("active", i === currentIndex);
        });
    }

    function goNextSlide() {
        currentIndex = (currentIndex + 1) % sliderImgs.length;
        updateSlider();
    }

    function goPrevSlide() {
        currentIndex = (currentIndex - 1 + sliderImgs.length) % sliderImgs.length;
        updateSlider();
    }

    // 버튼 클릭
    nextBtn.addEventListener("click", goNextSlide);
    prewBtn.addEventListener("click", goPrevSlide);

    // 페이저 클릭
    pagerItems.forEach((li, index) => {
        li.addEventListener("click", () => {
            currentIndex = index;
            updateSlider();
        });
    });

    updateSlider();

    // 🧲 스와이프 기능
    let startX = 0;
    let endX = 0;

    // 터치
    sliderImg.addEventListener("touchstart", (e) => {
        startX = e.touches[0].clientX;
    });

    sliderImg.addEventListener("touchend", (e) => {
        endX = e.changedTouches[0].clientX;
        handleSwipe();
    });

    // 마우스 (데스크탑 드래그 대응)
    sliderImg.addEventListener("mousedown", (e) => {
        startX = e.clientX;
    });

    sliderImg.addEventListener("mouseup", (e) => {
        endX = e.clientX;
        handleSwipe();
    });

    function handleSwipe() {
        const diff = startX - endX;
        if (Math.abs(diff) < 30) return; // 너무 짧으면 무시
        if (diff > 0) {
            // 왼쪽 → 오른쪽으로 드래그 (다음 이미지)
            goNextSlide();
        } else {
            // 오른쪽 → 왼쪽으로 드래그 (이전 이미지)
            goPrevSlide();
        }
    }
});
